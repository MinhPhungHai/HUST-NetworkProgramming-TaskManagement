Table users {
  user_id varchar(36) [primary key]
  username varchar(255) [not null, unique]
  email varchar(255) [not null, unique]
  phone_number varchar(20) [unique]
  password_hash varchar(255) [not null]
  is_verified boolean [default: false]
  created_at timestamp [default: `CURRENT_TIMESTAMP`]
  updated_at timestamp [default: `CURRENT_TIMESTAMP`]

  Note: 'User accounts and authentication'
}


Table otp_verifications {
  otp_id varchar(36) [primary key]
  user_id varchar(36) [ref: > users.user_id]
  email varchar(255)
  otp_code varchar(10) [not null]
  otp_type varchar(20) [not null, note: 'registration, login, password_reset']
  expires_at timestamp [not null]
  is_used boolean [default: false]
  created_at timestamp [default: `CURRENT_TIMESTAMP`]

  Note: 'OTP codes for email/phone verification'
}


Table user_contacts {
  contact_id varchar(36) [primary key]
  user_id varchar(36) [not null, ref: > users.user_id]
  contact_user_id varchar(36) [not null, ref: > users.user_id]
  status varchar(20) [default: 'pending', note: 'pending, accepted, blocked']
  created_at timestamp [default: `CURRENT_TIMESTAMP`]

  Note: 'Friend/contact relationships between users'

  Indexes {
    (user_id, contact_user_id) [unique]
  }
}


Table projects {
  project_id varchar(36) [primary key]
  project_name varchar(255) [not null]
  description text
  owner_id varchar(36) [not null, ref: > users.user_id]
  start_date date
  end_date date
  status varchar(20) [default: 'planning', note: 'planning, in_progress, completed, on_hold']
  created_at timestamp [default: `CURRENT_TIMESTAMP`]
  updated_at timestamp [default: `CURRENT_TIMESTAMP`]

  Note: 'Projects created by users'
}


Table project_members {
  member_id varchar(36) [primary key]
  project_id varchar(36) [not null, ref: > projects.project_id]
  user_id varchar(36) [not null, ref: > users.user_id]
  role varchar(20) [default: 'member', note: 'owner, admin, member, viewer']
  joined_at timestamp [default: `CURRENT_TIMESTAMP`]

  Note: 'Project membership and roles'

  Indexes {
    (project_id, user_id) [unique]
  }
}


Table tasks {
  task_id varchar(36) [primary key]
  project_id varchar(36) [not null, ref: > projects.project_id]
  task_name varchar(255) [not null]
  description text
  assigned_to varchar(36) [ref: > users.user_id]
  created_by varchar(36) [not null, ref: > users.user_id]
  status varchar(20) [default: 'todo', note: 'todo, in_progress, review, completed, blocked']
  priority varchar(20) [default: 'medium', note: 'low, medium, high, urgent']
  start_date date
  due_date date
  completed_at timestamp [null]
  created_at timestamp [default: `CURRENT_TIMESTAMP`]
  updated_at timestamp [default: `CURRENT_TIMESTAMP`]

  Note: 'Tasks within projects'
}


Table task_comments {
  comment_id varchar(36) [primary key]
  task_id varchar(36) [not null, ref: > tasks.task_id]
  user_id varchar(36) [not null, ref: > users.user_id]
  comment_content text [not null]
  created_at timestamp [default: `CURRENT_TIMESTAMP`]
  updated_at timestamp [default: `CURRENT_TIMESTAMP`]

  Note: 'Comments on tasks'
}


Table project_chat_log {
  chat_id varchar(36) [primary key]
  project_id varchar(36) [not null, ref: > projects.project_id]
  sender_id varchar(36) [not null, ref: > users.user_id]
  message_content text [not null]
  created_at timestamp [default: `CURRENT_TIMESTAMP`]

  Note: 'Chat messages within projects'
}


Table file_attachments {
  file_id varchar(36) [primary key]
  task_id varchar(36) [not null, ref: > tasks.task_id]
  file_name varchar(255) [not null]
  file_path varchar(500) [not null]
  file_size bigint [not null]
  file_type varchar(100)
  uploaded_by varchar(36) [not null, ref: > users.user_id]
  created_at timestamp [default: `CURRENT_TIMESTAMP`]

  Note: 'File attachments for tasks'
}


Table notifications {
  notification_id varchar(36) [primary key]
  user_id varchar(36) [not null, ref: > users.user_id]
  notification_type varchar(50) [not null, note: 'task_assigned, project_invite, comment_added, task_updated, etc.']
  related_entity_type varchar(50) [note: 'task, project, comment, etc.']
  related_entity_id varchar(36)
  message text [not null]
  is_read boolean [default: false]
  created_at timestamp [default: `CURRENT_TIMESTAMP`]

  Note: 'User notifications'
}
